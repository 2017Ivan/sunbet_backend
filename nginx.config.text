server {

    # Tunaiambia Nginx ipokee request yoyote inayokuja kupitia IP ya server yako
    server_name boombet365.com www.boombet365.com;

    root /var/www/boombet_frontend;
    index index.html;

    # Frontend Routing (Vue/Vite)
    location / {
        try_files $uri $uri/ /index.html;
    }

    # Proxy kwenda kwenye PM2 Backend (Node/Python)
    location /api/ {
        proxy_pass http://127.0.0.1:5000/api/;
        proxy_http_version 1.1;

        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    listen [::]:443 ssl ipv6only=on; # managed by Certbot
    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/boombet365.com/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/boombet365.com/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot


}
